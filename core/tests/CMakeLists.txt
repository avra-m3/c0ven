# The core's own tests (docs/TESTING.md): plain executables under ctest. Tests that need a card carry the label
# "gpu"; tests that need the model skip (exit code 77) when FLASH_MODEL_DIR does not hold it.

function(engine_test name)
  cmake_parse_arguments(T "GPU" "" "SOURCES;ARGS" ${ARGN})
  add_executable(${name} ${T_SOURCES})
  target_link_libraries(${name} PRIVATE engine_core engine_ggml engine_warnings)
  add_test(NAME ${name} COMMAND ${name} ${T_ARGS})
  set_tests_properties(${name} PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 600)
  if(T_GPU)
    set_tests_properties(${name} PROPERTIES LABELS gpu)
    get_property(kernels GLOBAL PROPERTY ENGINE_KERNEL_TARGETS)
    add_dependencies(${name} ${kernels})
  endif()
endfunction()
